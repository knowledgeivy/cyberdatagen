% \text{F1} = \frac{2 \cdot TP}{2 \cdot TP + FP + FN}
% \]

% \paragraph{Supporting Metrics:}
% \begin{itemize}
%     \item \textbf{Accuracy:} $(TP + TN) / (TP + TN + FP + FN)$
%     \item \textbf{Precision:} $TP / (TP + FP)$ - false positive control
%     \item \textbf{Recall:} $TP / (TP + FN)$ - detection sensitivity
%     \item \textbf{AUC-ROC:} Area under ROC curve, threshold-independent discrimination
%     \item \textbf{Balanced Accuracy:} $\frac{1}{2}\left(\frac{TP}{TP+FN} + \frac{TN}{TN+FP}\right)$ - class-imbalance robust
% \end{itemize}
