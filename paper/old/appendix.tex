\section{Appendices}
\label{sec:appendix}

This appendix provides the mathematical formulations for the comprehensive embedding metrics introduced in our evaluation framework.

\subsection{Embedding Space Quality Metrics}

\subsubsection{Centroid Distance}
The centroid distance measures the Euclidean displacement between synthetic strategy centroids and real malicious data centroid:

\begin{equation}
d_{centroid}(S_i) = \|\mathbf{c}_{S_i} - \mathbf{c}_{real}\|_2
\end{equation}

where $\mathbf{c}_{S_i} = \frac{1}{|S_i|} \sum_{\mathbf{x} \in S_i} \mathbf{x}$ represents the centroid of synthetic strategy $S_i$, and $\mathbf{c}_{real} = \frac{1}{|R|} \sum_{\mathbf{x} \in R} \mathbf{x}$ represents the centroid of real malicious data $R$.

\subsubsection{Mean Cosine Similarity}
The mean cosine similarity quantifies semantic preservation by measuring the average maximum cosine similarity between synthetic samples and real malicious samples:

\begin{equation}
\text{MCS}(S_i) = \frac{1}{|S_i|} \sum_{\mathbf{s} \in S_i} \max_{\mathbf{r} \in R} \frac{\mathbf{s} \cdot \mathbf{r}}{\|\mathbf{s}\|_2 \|\mathbf{r}\|_2}
\end{equation}

where $\mathbf{s}$ represents a synthetic sample and $\mathbf{r}$ represents a real malicious sample.

\subsubsection{Co-clustering Rate}
The co-clustering rate measures structural similarity by calculating the frequency of synthetic samples clustering with real malicious samples:

\begin{equation}
\text{CCR}(S_i) = \frac{1}{|S_i| \cdot |R|} \sum_{\mathbf{s} \in S_i} \sum_{\mathbf{r} \in R} \mathbb{I}[C(\mathbf{s}) = C(\mathbf{r})]
\end{equation}

where $C(\mathbf{x})$ denotes the cluster assignment of sample $\mathbf{x}$ under optimal K-means configuration, and $\mathbb{I}[\cdot]$ is the indicator function.

\subsubsection{Semantic Drift Score}
The semantic drift score quantifies normalized deviation from real malicious distribution:

\begin{equation}
\text{SDS}(S_i) = \frac{\frac{1}{|S_i|} \sum_{\mathbf{s} \in S_i} \|\mathbf{s} - \mathbf{c}_{real}\|_2}{\sigma_{real}}
\end{equation}

where $\sigma_{real} = \sqrt{\frac{1}{|R|} \sum_{\mathbf{r} \in R} \|\mathbf{r} - \mathbf{c}_{real}\|_2^2}$ represents the standard deviation of real sample distances from their centroid.

\subsubsection{Intra-strategy Diversity}
The intra-strategy diversity measures internal variation within each generation strategy:

\begin{equation}
\text{ISD}(S_i) = \frac{2}{|S_i|(|S_i|-1)} \sum_{j=1}^{|S_i|-1} \sum_{k=j+1}^{|S_i|} \|\mathbf{s}_j - \mathbf{s}_k\|_2
\end{equation}

where $\mathbf{s}_j$ and $\mathbf{s}_k$ represent distinct samples within strategy $S_i$.

\subsubsection{Distribution Coverage}
The distribution coverage evaluates how effectively synthetic samples cover the real malicious data distribution:

\begin{equation}
\text{DC}(S_i) = \frac{1}{|R|} \sum_{\mathbf{r} \in R} \frac{1}{1 + \min_{\mathbf{s} \in S_i} \|\mathbf{r} - \mathbf{s}\|_2}
\end{equation}

This metric assigns higher coverage scores when synthetic samples are positioned close to real samples in embedding space.
